```gdscript
var ads := Intermission.new()

func _ready() -> void:
	add_child(ads)
	ads.ad_closed.connect(_on_ad_closed)
	ads.initialize("YOUR_APP_KEY", OS.is_debug_build())
	ads.load_ad(Intermission.Format.REWARDED, "YOUR_REWARDED_AD_UNIT_ID")
```
